\subsection{Pseudo-code de l'algorithme SAC}

\begin{algorithm}[H]
    \caption{Soft Actor-Critic}
    \label{alg:sac_training}

    \vspace{10pt}

    \textbf{inputs :}

    \hspace{15pt} $\theta$ : paramètres initiaux de la politique

    \hspace{15pt} $\phi_1$, $\phi_2$ : paramètres initiaux des fonctions de valeurs d'action

    \hspace{15pt} $\mathcal{D}$ : replay buffer vide

    \hspace{15pt} $\eta_{\text{critic}}$, $\eta_{\text{actor}}$ : taux d'apprentissage

    \vspace{10pt}


    \textbf{initialisation :}
    $\phi_{1, \text{target}} \leftarrow \phi_1$,
    $\phi_{2, \text{target}} \leftarrow \phi_2$

    \vspace{10pt}

    \textcolor{PineGreen}{\texttt{\# On applique la procédure jusqu'à convergence... }}

    \For{\small{\texttt{chaque itération}}}{

        \vspace{10pt}

        \For{\small{\texttt{chaque étape dans l'environnement}}}{

            \textcolor{PineGreen}{\texttt{\# On observe $\mathbf{s}$ puis on exécute $\mathbf{a}$ dans l'environnement }}

            $\mathbf{a} \sim \pi_\theta(\cdot \mid \mathbf{s})$

            $\mathbf{s}' \sim P(\cdot \mid \mathbf{s}, \; \mathbf{a})$

            $r \leftarrow R(\mathbf{s}, \; \mathbf{a})$

            \textcolor{PineGreen}{\texttt{\# On stocke les transitions dans le replay buffer }}

            $\mathcal{D} \leftarrow \mathcal{D} \; \cup \; (\mathbf{s}, \;  \mathbf{a},\;  r,\;  \mathbf{s}', \; d)$

            \textcolor{PineGreen}{\texttt{\# Si $\mathbf{s}'$ est terminal, on recommence}}

            \textbf{if} $d = 1$ \textbf{then} réinitialiser l'environnement
        }

        \vspace{10pt}

        \For{\small{\texttt{chaque étape de mise à jour}}}{
            \textcolor{PineGreen}{\texttt{\# Échantillonner un batch de transitions aléatoirement}}

            $B \sim \mathcal{D}$

            \textcolor{PineGreen}{\texttt{\# Calculer les targets des fonctions $Q$}}

            $\tilde{\mathbf{a}}' \sim \pi_\theta(\cdot \mid \mathbf{s}')$

            $
                y \leftarrow r + \gamma(1 - d) \left( \min_{i \in \{1,\; 2\}} Q_{\phi_{i, \text{target}}} (\mathbf{s}', \; \tilde{\mathbf{a}}') - \alpha \log \pi(\tilde{\mathbf{a}}' \mid \mathbf{s}')\right)
            $

            \textcolor{PineGreen}{\texttt{\# Mettre à jour $Q_{\phi_1}$ et $Q_{\phi_2}$ par descente de gradient, avec :}}

            $
                \phi_i
                \leftarrow
                \phi_i - \eta_{\text{critic}}
                \nabla_{\phi_i} \frac{1}{|B|} \sum_{(\mathbf{s},\;  \mathbf{a}, \; r, \; \mathbf{s}',\;  d) \in B} (Q_{\phi_i}(\mathbf{s}, \; \mathbf{a}) - y)^2
            $

            \textcolor{PineGreen}{\texttt{\# Mettre à jour $\pi_\theta$ par montée de gradient, avec :}}

            $
                \varepsilon \sim \mathcal{N}(0, \; I)
            $

            $
                \tilde{\textbf{a}}_\theta(\textbf{s}, \varepsilon) \leftarrow \tanh( \mu _\theta (\textbf{s}) + \sigma _\theta(\textbf{s})\odot \varepsilon)
            $

            $
                \theta \leftarrow
                \theta + \eta_{\text{actor}}
                \nabla_{\theta} \frac{1}{|B|} \sum_{\mathbf{s} \in B} \left( \min_{i\in\{1, 2\}}Q_{\phi_i}(\mathbf{s}, \; \tilde{\mathbf{a}}_\theta ) -  \alpha \log \pi_\theta(\tilde{\mathbf{a}}_\theta  \mid \mathbf{s})\right)
            $

            \textcolor{PineGreen}{\texttt{\# Mettre à jour les target networks :}}

            $
                \phi_{1, \text{target}} \leftarrow (1 - \lambda) \phi_{1, \text{target}} + \lambda\phi_{1}
            $

            $
                \phi_{2, \text{target}} \leftarrow (1 - \lambda) \phi_{2, \text{target}} + \lambda\phi_{2}
            $

        }
    }
\end{algorithm}
